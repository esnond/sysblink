# написание патчей для sysblink

патч это один файл c в каталоге patches. установщик вставляет его в
engine.c и компилирует результат, поэтому патч это часть большей программы,
а не самостоятельная. это руководство описывает контракт, подручные
средства и один разобранный пример.

## контракт

* патч определяет ровно одну публичную точку входа: int get_led_value(void)
* движок вызывает ее раз за тик и ждет уровень от 0 до 100
* 0 значит темно, 100 значит полная яркость, масштабирование под максимум
  диода делает движок, не делайте это дважды
* функция должна возвращаться быстро, сон внутри тормозит всего демона
* состояние храните в статических переменных, функция вызывается снова
* все символы кроме точки входа должны быть static

## что может использовать патч

движок определяет это до кода патча:

* BLINK_HZ: частота тика в герцах, переводите бюджеты за секунду в бюджеты
  за тик делением
* BLINK_PROC_ROOT и BLINK_SYS_ROOT: корни /proc и /sys в виде строк,
  пристраивайте свои пути к ним, тесты направляют их на фикстуры
* BLINK_DISK_DEVICE и BLINK_NET_IFACE: подсказки выбора из sysblink.conf в
  виде строк, пустое значение значит поведение по умолчанию, проверяйте
  их, если патч следит за одним устройством или интерфейсом из многих
* LED_COUNT, LED_PATHS, LED_MAXS: настройка диодов, патчу они не нужны,
  один патч обслуживает все диоды демона
* SLEEP_US: внутренности движка, обычно не трогаются

общие пороги лежат в patches/config.h, подключайте его где нужно:

    #include "config.h"

подключайте заголовки, которые используете, движок ничего не подключает
за патч, кроме собственных нужд.

## измерение скорости

патчу скорости нужны два замера и защита от самого первого вызова, у
которого нет предыдущего для сравнения:

    static uint64_t prev_total;
    static int primed;

    total = ...;
    if (!primed) {
        primed = 1;
        prev_total = total;
        return 0;
    }
    diff = total > prev_total ? total - prev_total : 0;
    prev_total = total;

еще два правила держат числа честными:

* счетчики могут уйти назад после сна машины, отрицательную разницу
  считайте нулем
* переводите бюджет из config.h в бюджет за тик через BLINK_HZ, например
  budget = NET_BYTES_PER_SEC / BLINK_HZ, и сравнивайте разницу с ним

## компиляторный шлюз

проект собирается так:

    cc -std=c99 -pedantic -Wall -Wextra -Werror -O2

поэтому патч должен быть чистым iso c99 плюс маленькое posix подмножество,
которое движок уже включает: без gnu расширений, без новых библиотек,
с типами фиксированной ширины из stdint.h для счетчиков, с макросами
форматов из inttypes.h для scanf и printf, с явным void в списке
параметров точки входа.

## полный пример

patches/swap.c это законченный маленький патч, он показывает открытие
файла из proc, разбор двух полей через sscanf и шаблон со статическим
состоянием:

    /*
     * sysblink patch: swap usage
     * example patch from docs/patches.md, reads SwapTotal and SwapFree
     * from /proc/meminfo and returns the share of swap in use
     */

    #include <inttypes.h>
    #include <stdio.h>
    #include <string.h>

    #define SWAP_LINE_SIZE     256
    #define SWAP_LABEL_SIZE    32
    #define SWAP_FIELDS_NEEDED 2

    int get_led_value(void)
    {
        uint64_t total = 0, free_pages = 0, value;
        char label[SWAP_LABEL_SIZE];
        char line[SWAP_LINE_SIZE];
        int found = 0;
        FILE *fp;

        fp = fopen("/proc/meminfo", "r");
        if (fp == NULL)
            return 0;
        while (found < SWAP_FIELDS_NEEDED && fgets(line, sizeof(line), fp) != NULL) {
            if (sscanf(line, "%31s %" SCNu64, label, &value) != 2)
                continue;
            if (strcmp(label, "SwapTotal:") == 0) {
                total = value;
                found++;
            } else if (strcmp(label, "SwapFree:") == 0) {
                free_pages = value;
                found++;
            }
        }
        fclose(fp);

        if (found < SWAP_FIELDS_NEEDED || total == 0)
            return 0;
        return (int)(100 * (total - free_pages) / total);
    }

чтобы добавить свой патч, сохраните листинг как patches/имя.c и запустите
make test. если свопа на машине нет, патч отвечает нулем и диод темный.

## проверка патча

    make test

собирает ваш патч в движок со строгими флагами и гоняет два вида headless
сессий. живые используют настоящий /proc и /sys и проверяют только то,
что числа вменяемые. проверочные направляют движок на tests/fixtures и
ждут точных значений, поэтому если ваш патч читает источник, для которого
уже есть фикстура, его число зафиксировано. у патчей со временем вместо
этого ищется яркая фаза. посмотреть готовый патч вживую без установки:

    python3 preview.py breathe --hz 60
    watch cat build/led_breathe.txt

## чеклист

* файл определяет get_led_value и ничего публичного сверх нее
* помощники и состояние static, включения минимальные и присутствуют
* осмысленные пороги живут в patches/config.h или как именованные макросы
  в начале патча, единица измерения указана в комментарии
* комментарии объясняют почему, строчными буквами, без тире
* make test проходит с вашим патчем на месте
